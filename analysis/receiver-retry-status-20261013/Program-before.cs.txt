using IMouse2.Kernel;

var tests = new (string Name, Func<Task> Run)[]
{
    ("cancelled startup never launches and reports stopped", async () => {
        var starts = 0; using var cancel = new CancellationTokenSource(); cancel.Cancel();
        using var s = new AirplaySupervisor(() => { starts++; return new Fake(); });
        await s.RunAsync(cancel.Token);
        Check(starts == 0 && s.Info.State == "stopped", "cancelled start state");
    }),
    ("running cancellation disposes once and reports stopped", async () => {
        var fake = new Fake(); using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => fake);
        var run = s.RunAsync(cancel.Token); cancel.Cancel(); await run.WaitAsync(TimeSpan.FromSeconds(2));
        Check(fake.Disposals == 1 && s.Info is { State: "stopped", Pid: 0 }, "running cancellation cleanup");
    }),
    ("repeated Dispose never disposes a child twice", async () => {
        var fake = new Fake(); var s = new AirplaySupervisor(() => fake);
        var run = s.RunAsync(CancellationToken.None); s.Dispose(); s.Dispose();
        await run.WaitAsync(TimeSpan.FromSeconds(2));
        Check(fake.Disposals == 1, $"dispose count {fake.Disposals}");
    }),
    ("startup exceptions exhaust a finite budget", async () => {
        int starts = 0; using var s = new AirplaySupervisor(() => { starts++; throw new IOException("fake start"); }, 2, TimeSpan.Zero);
        await s.RunAsync(CancellationToken.None);
        Check(starts == 3 && s.Info is { State: "failed", Pid: 0 }, "start budget");
    }),
    ("zero retry budget reports no replacement attempts", async () => {
        int starts = 0;
        using var s = new AirplaySupervisor(() => { starts++; throw new IOException("launch fixture"); }, 0, TimeSpan.Zero);
        await s.RunAsync(CancellationToken.None);
        Check(starts == 1 && s.Info is { State: "failed", Restarts: 0 }, $"one launch, reported retries {s.Info.Restarts}");
    }),
    ("retry exhaustion reports actual attempts and the latest startup failure", async () => {
        int starts = 0; var counts = new List<int>(); AirplaySupervisor? s = null;
        s = new AirplaySupervisor(() => { counts.Add(s!.Info.Restarts); throw new IOException("launch fixture " + ++starts); }, 2, TimeSpan.Zero);
        using (s) {
            await s.RunAsync(CancellationToken.None);
            Check(counts.SequenceEqual(new[] { 0, 1, 2 }), "counter must be visible when an attempt starts");
            Check(s.Info.Restarts == 2, $"two replacement attempts, reported {s.Info.Restarts}");
            Check(s.Info.Message.Contains("launch fixture 3") && !s.Info.Message.Contains("launch fixture 1"), "latest startup cause lost: " + s.Info.Message);
        }
    }),
    ("startup failure stays visible while waiting without consuming a retry", async () => {
        int starts = 0; using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => { starts++; throw new IOException("missing dependency fixture"); }, 2, TimeSpan.FromSeconds(10));
        var run = s.RunAsync(cancel.Token);
        try {
            await Until(() => s.Info.State == "restarting");
            Check(s.Info.Message.Contains("missing dependency fixture"), "backoff hides startup reason: " + s.Info.Message);
            Check(s.Info.Restarts == 0, "backoff is not a launch");
        } finally { cancel.Cancel(); await run.WaitAsync(TimeSpan.FromSeconds(2)); }
        Check(starts == 1 && s.Info is { State: "stopped", Restarts: 0 }, "cancelled backoff consumed a retry");
    }),
    ("wait failure retains its cause after retry exhaustion", async () => {
        using var s = new AirplaySupervisor(() => new Fake { WaitFailure = new IOException("exit monitor fixture") }, 0, TimeSpan.Zero);
        await s.RunAsync(CancellationToken.None);
        Check(s.Info.State == "failed" && s.Info.Message.Contains("exit monitor fixture"), "wait failure cause lost: " + s.Info.Message);
    }),
    ("disposing an exhausted supervisor preserves its failure record", async () => {
        var s = new AirplaySupervisor(() => throw new IOException("launch fixture"), 0, TimeSpan.Zero);
        await s.RunAsync(CancellationToken.None);
        var failed = s.Info; s.Dispose(); s.Dispose();
        Check(s.Info == failed && failed.State == "failed", "Dispose erased exhausted failure state");
    }),
    ("cleanup failure retains the last receiver identity", async () => {
        var fake = new Fake { DisposeFailure = true }; fake.Exit.TrySetResult();
        using var s = new AirplaySupervisor(() => fake, 2, TimeSpan.Zero);
        await s.RunAsync(CancellationToken.None);
        Check(s.Info is { State: "failed", Pid: 123, Restarts: 0 } && s.Info.Message.Contains("123"), "unconfirmed cleanup lost its last known PID");
        s.Dispose(); Check(s.Info.Pid == 123, "shutdown erased unconfirmed receiver identity");
    }),
    ("a recovered receiver clears the previous startup error", async () => {
        int starts = 0; var healthy = new Fake(); using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => ++starts == 1 ? throw new IOException("old fixture") : healthy, 1, TimeSpan.Zero);
        var run = s.RunAsync(cancel.Token);
        try {
            await Until(() => s.Info.State == "running");
            Check(s.Info is { Restarts: 1, Message: "", Pid: 123 }, "old error leaked into healthy state");
        } finally { cancel.Cancel(); await run.WaitAsync(TimeSpan.FromSeconds(2)); }
    }),
    ("cleanup failure during cancelled startup remains visible", async () => {
        var fake = new Fake { DisposeFailure = true }; AirplaySupervisor? s = null;
        s = new AirplaySupervisor(() => { s!.Dispose(); return fake; });
        using (s) {
            await s.RunAsync(CancellationToken.None).WaitAsync(TimeSpan.FromSeconds(2));
            Check(fake.Disposals == 1 && s.Info is { State: "failed", Pid: 123, Restarts: 0 }, "unpublished child cleanup failure lost");
        }
    }),
    ("wait failure cleans up and recovers", async () => {
        var first = new Fake { WaitFailure = new IOException("fake wait") }; var second = new Fake(); int starts = 0;
        using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => ++starts == 1 ? first : second, 1, TimeSpan.Zero);
        var run = s.RunAsync(cancel.Token);
        try { await Until(() => starts == 2 || run.IsCompleted); Check(starts == 2 && first.Disposals == 1, "wait failure must retry"); }
        finally { cancel.Cancel(); try { await run.WaitAsync(TimeSpan.FromSeconds(2)); } catch (IOException) { } }
        Check(second.Disposals == 1, "recovered receiver cleanup");
    }),
    ("unrelated cancellation exception is a failure, not shutdown", async () => {
        int starts = 0; var last = new Fake(); using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => ++starts == 1 ? new Fake { WaitFailure = new OperationCanceledException() } : last, 1, TimeSpan.Zero);
        var run = s.RunAsync(cancel.Token);
        try { await Until(() => starts == 2 || run.IsCompleted); Check(starts == 2, "unrelated cancellation swallowed"); }
        finally { cancel.Cancel(); await run.WaitAsync(TimeSpan.FromSeconds(2)); }
    }),
    ("cancel during backoff prevents replacement", async () => {
        int starts = 0; var dead = new Fake(); dead.Exit.TrySetResult(); using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => { starts++; return dead; }, 3, TimeSpan.FromSeconds(10));
        var run = s.RunAsync(cancel.Token); await Until(() => s.Info.State == "restarting");
        cancel.Cancel(); await run.WaitAsync(TimeSpan.FromSeconds(2));
        Check(starts == 1 && s.Info.State == "stopped", "backoff cancellation");
    }),
    ("negative delay is refused before starting", () => {
        bool refused = false;
        try { using var s = new AirplaySupervisor(() => new Fake(), delay: TimeSpan.FromMilliseconds(-1)); }
        catch (ArgumentOutOfRangeException) { refused = true; }
        Check(refused, "negative delay accepted"); return Task.CompletedTask;
    }),
    ("one failed receiver leaves a healthy peer running", async () => {
        int starts = 0; var healthy = new Fake(); using var cancel = new CancellationTokenSource();
        using var peer = new AirplaySupervisor(() => healthy);
        using var failed = new AirplaySupervisor(() => { starts++; var x = new Fake(); x.Exit.TrySetResult(); return x; }, 1, TimeSpan.Zero);
        var live = peer.RunAsync(cancel.Token); await failed.RunAsync(CancellationToken.None);
        Check(starts == 2 && failed.Info.State == "failed" && peer.Info.State == "running" && healthy.Disposals == 0, "peer isolation");
        cancel.Cancel(); await live;
    }),
    ("cleanup failure stops replacements", async () => {
        int starts = 0; var fake = new Fake { DisposeFailure = true }; fake.Exit.TrySetResult();
        using var s = new AirplaySupervisor(() => { starts++; return fake; }, 2, TimeSpan.Zero);
        await s.RunAsync(CancellationToken.None);
        Check(starts == 1 && s.Info.State == "failed" && s.Info.Message.Contains("cleanup", StringComparison.OrdinalIgnoreCase), "cleanup failure must block replacement");
        s.Dispose(); Check(s.Info.State == "failed", "Dispose must not hide unconfirmed cleanup");
    }),
    ("second run cannot start a second receiver", async () => {
        int starts = 0; using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => { starts++; return new Fake(); });
        var first = s.RunAsync(cancel.Token); bool refused = false;
        try { await s.RunAsync(cancel.Token).WaitAsync(TimeSpan.FromMilliseconds(100)); }
        catch (InvalidOperationException) { refused = true; }
        finally { cancel.Cancel(); await first.WaitAsync(TimeSpan.FromSeconds(2)); }
        Check(refused && starts == 1, "duplicate run allowed");
    }),
    ("logging warnings do not replace receiver lifecycle state", async () => {
        var fake = new Fake { DiagnosticWarning = "Receiver log unavailable: fixture" };
        using var cancel = new CancellationTokenSource();
        using var s = new AirplaySupervisor(() => fake);
        var run = s.RunAsync(cancel.Token);
        Check(s.Info.State == "running" && s.Info.Message.Contains("log unavailable"), "logging failure not reported");
        fake.DiagnosticWarning = null;
        Check(s.Info.Message == "" && s.Info.Pid == 123, "logging recovery not reported");
        cancel.Cancel(); await run;
    }),
    ("receiver logs rotate at their byte budget", () => InLogFolder(folder => {
        string path = Path.Combine(folder,"receiver.log");
        var log = new ReceiverLog(path,512);
        for(int i=0;i<30;i++) log.Append("line-"+i+" "+new string('x',30));
        Check(log.Error is null && File.Exists(path+".1"), "rotation failed: " + log.Error);
        Check(new FileInfo(path).Length <= 512 && new FileInfo(path+".1").Length <= 512, "log size exceeded");
        Check(File.ReadAllText(path).Contains("line-29") && Directory.GetFiles(folder).Length==2, "latest output or retention wrong");
    })),
    ("receiver logs truncate huge Unicode lines within byte budget", () => InLogFolder(folder => {
        string path=Path.Combine(folder,"unicode.log"); var log=new ReceiverLog(path,512);
        log.Append(string.Concat(Enumerable.Repeat("未分组😀",10000)));
        string text=File.ReadAllText(path);
        Check(log.Error is null && new FileInfo(path).Length<=512 && text.Contains("[truncated]"),"Unicode bound failed");
    })),
    ("concurrent stdout and stderr keep complete lines", async () => {
        string folder=NewLogFolder();
        try {
            string path=Path.Combine(folder,"parallel.log"); var log=new ReceiverLog(path);
            await Task.WhenAll(Enumerable.Range(0,100).Select(i=>Task.Run(()=>log.Append("record-"+i))));
            var lines=File.ReadAllLines(path);
            Check(log.Error is null && lines.Length==100 && lines.Select(x=>x[(x.IndexOf("record-"))..]).Distinct().Count()==100,"concurrent line loss");
        } finally { Directory.Delete(folder,true); }
    }),
    ("unwritable log reports failure and later recovers", () => InLogFolder(folder => {
        string missing=Path.Combine(folder,"not-yet-created");
        string path=Path.Combine(missing,"receiver.log"); var log=new ReceiverLog(path);
        log.Append("first"); Check(log.Error?.Contains("log unavailable")==true,"file failure escaped or hidden");
        Directory.CreateDirectory(missing); log.Append("second");
        Check(log.Error is null && File.ReadAllText(path).Contains("second"),"logging did not recover");
    })),
    ("failed rotation does not grow or discard active log", () => InLogFolder(folder => {
        string path=Path.Combine(folder,"blocked.log"); var log=new ReceiverLog(path,256);
        File.WriteAllText(path,new string('a',250)); Directory.CreateDirectory(path+".1");
        log.Append("line");
        Check(log.Error is not null && File.ReadAllText(path)==new string('a',250),"rotation error changed active log");
        Directory.Delete(path+".1"); log.Append("retry");
        Check(log.Error is null && File.ReadAllText(path).Contains("retry"),"rotation retry failed: " + log.Error);
    })),
    ("pre-existing oversized log is retained for one generation", () => InLogFolder(folder => {
        string path=Path.Combine(folder,"legacy.log"); var log=new ReceiverLog(path,512);
        File.WriteAllText(path,new string('a',2000)); log.Append("new generation");
        Check(log.Error is null,"legacy rotation failed: " + log.Error);
        Check(File.ReadAllText(path+".1")==new string('a',2000),"legacy evidence truncated");
        Check(new FileInfo(path).Length<=512 && log.Error is null,"new generation not bounded");
    })),
    ("empty callback does not create a log", () => InLogFolder(folder => {
        string path=Path.Combine(folder,"empty.log"); var log=new ReceiverLog(path);
        log.Append(null); Check(!File.Exists(path) && log.Error is null,"EOF wrote a line");
    })),
    ("dispose inside startup cannot publish a running child", async () => {
        var fake = new Fake(); AirplaySupervisor? s = null;
        s = new AirplaySupervisor(() => { s!.Dispose(); return fake; });
        await s.RunAsync(CancellationToken.None).WaitAsync(TimeSpan.FromSeconds(2));
        Check(s.Info.State == "stopped" && fake.Disposals == 1, "startup cancellation race"); s.Dispose();
    })
};
int passed = 0;
foreach (var (name, run) in tests)
{
    try { await run(); Console.WriteLine("PASS " + name); passed++; }
    catch (Exception ex) { Console.WriteLine("FAIL " + name + ": " + ex.Message); }
}
Console.WriteLine($"{passed}/{tests.Length} recovery checks passed; fake handles only; no sockets, HID, receivers or UI.");
return passed == tests.Length ? 0 : 1;

static void Check(bool value, string message) { if (!value) throw new Exception(message); }
static string NewLogFolder() { string path=Path.Combine(Path.GetTempPath(),"imouse-log-test-"+Guid.NewGuid().ToString("N")); Directory.CreateDirectory(path); return path; }
static Task InLogFolder(Action<string> test)
{
    string folder=NewLogFolder();
    try { test(folder); } finally { Directory.Delete(folder,true); }
    return Task.CompletedTask;
}
static async Task Until(Func<bool> ready)
{
    using var deadline = new CancellationTokenSource(TimeSpan.FromSeconds(2));
    while (!ready()) await Task.Delay(1, deadline.Token);
}
sealed class Fake : IReceiverHandle
{
    public int Id => 123;
    public TaskCompletionSource Exit { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
    public int Disposals;
    public Exception? WaitFailure;
    public bool DisposeFailure;
    public string? DiagnosticWarning { get; set; }
    public Task WaitForExitAsync(CancellationToken ct) => WaitFailure is {} ex ? Task.FromException(ex) : Exit.Task.WaitAsync(ct);
    public void Dispose() { Interlocked.Increment(ref Disposals); if (DisposeFailure) throw new IOException("fake cleanup"); Exit.TrySetResult(); }
}
